```
document.addEventListener("DOMContentLoaded", () => {
  const form = document.getElementById("kykloi-decrypt-form");
  if (!form) return;

  const passwordInput = document.getElementById("kykloi-password");
  const submitButton = form.querySelector('button[type="submit"]');
  const errorMsg = document.getElementById("kykloi-decrypt-error");
  const lockContainer = document.getElementById("kykloi-lock-container");
  const contentContainer = document.getElementById("kykloi-decrypted-content");
  const payloadEl = document.getElementById("kykloi-encrypted-payload");

  if (!payloadEl) return;

  if (
    typeof window.crypto === "undefined" ||
    typeof window.crypto.subtle === "undefined"
  ) {
    console.error(
      "[Kykloi] WebCrypto API is unavailable in this browsing context."
    );
    if (errorMsg) {
      errorMsg.textContent =
        "Decryption unavailable: WebCrypto is disabled by your browser in this context.";
      errorMsg.style.display = "block";
    }
    if (passwordInput) passwordInput.disabled = true;
    if (submitButton) submitButton.disabled = true;
    return;
  }

  function renderDecryptedMath(container) {
    if (typeof katex === "undefined") {
      console.warn("[Kykloi] KaTeX engine not available in global scope.");
      return;
    }

    const mathElements = container.querySelectorAll(".math");
    mathElements.forEach((el) => {
      let tex = el.textContent.trim();

      if (tex.startsWith("\\(") && tex.endsWith("\\)")) {
        tex = tex.slice(2, -2).trim();
      } else if (tex.startsWith("\\[") && tex.endsWith("\\]")) {
        tex = tex.slice(2, -2).trim();
      } else if (tex.startsWith("$$") && tex.endsWith("$$")) {
        tex = tex.slice(2, -2).trim();
      } else if (tex.startsWith("$") && tex.endsWith("$")) {
        tex = tex.slice(1, -1).trim();
      }

      const isDisplay =
        el.classList.contains("display") ||
        el.classList.contains("math-display") ||
        el.tagName === "DIV";

      try {
        katex.render(tex, el, {
          displayMode: isDisplay,
          throwOnError: false,
        });
      } catch (kErr) {
        console.error("[Kykloi KaTeX Error]:", kErr);
      }
    });
  }

  function base64ToUint8Array(b64) {
    const binaryString = atob(b64);
    const len = binaryString.length;
    const bytes = new Uint8Array(len);

    for (let i = 0; i < len; i++) {
      bytes[i] = binaryString.charCodeAt(i);
    }
    return bytes;
  }

  async function deriveAesKey(password, salt) {
    const encoder = new TextEncoder();
    const keyMaterial = await window.crypto.subtle.importKey(
      "raw",
      encoder.encode(password),
      { name: "PBKDF2" },
      false,
      ["deriveKey"]
    );

    return await window.crypto.subtle.deriveKey(
      {
        name: "PBKDF2",
        salt: salt,
        iterations: 100000,
        hash: "SHA-256",
      },
      keyMaterial,
      { name: "AES-GCM", length: 256 },
      false,
      ["decrypt"]
    );
  }

  async function decryptEnvelope(password, rawText) {
    let base64Payload = rawText.trim();
    if (base64Payload.startsWith('"') && base64Payload.endsWith('"')) {
      try {
        base64Payload = JSON.parse(base64Payload);
      } catch (e) {
        console.warn("[Kykloi] JSON parse fallback used:", e);
      }
    }

    const bytes = base64ToUint8Array(base64Payload);
    if (bytes.length < 28) {
      throw new Error("Payload too short to contain Salt + IV headers.");
    }

    const salt = bytes.subarray(0, 16);
    const iv = bytes.subarray(16, 28);
    const ciphertext = bytes.subarray(28);

    const key = await deriveAesKey(password, salt);

    const decryptedBuffer = await window.crypto.subtle.decrypt(
      { name: "AES-GCM", iv: iv },
      key,
      ciphertext
    );

    return new TextDecoder().decode(decryptedBuffer);
  }

  function getMimeTypeFromUrl(url) {
    const cleanUrl = url.split(/[?#]/)[0].replace(/\.enc$/i, "");
    const ext = cleanUrl.split(".").pop().toLowerCase();
    const mimeMap = {
      jpg: "image/jpeg",
      jpeg: "image/jpeg",
      png: "image/png",
      gif: "image/gif",
      webp: "image/webp",
      svg: "image/svg+xml",
      bmp: "image/bmp",
      ico: "image/x-icon",
    };
    return mimeMap[ext] || "application/octet-stream";
  }

  async function decryptBinaryBlob(password, arrayBuffer) {
    const bytes = new Uint8Array(arrayBuffer);
    if (bytes.length < 28) {
      throw new Error("Encrypted blob too short.");
    }

    const salt = bytes.subarray(0, 16);
    const iv = bytes.subarray(16, 28);
    const ciphertext = bytes.subarray(28);

    const key = await deriveAesKey(password, salt);

    return await window.crypto.subtle.decrypt(
      { name: "AES-GCM", iv: iv },
      key,
      ciphertext
    );
  }

  async function hydrateEncryptedImages(container, password) {
    const encryptedImgs = container.querySelectorAll(
      'img[data-encrypted-media="true"], img[src$=".enc"]'
    );

    if (encryptedImgs.length === 0) return;

    for (const img of encryptedImgs) {
      const srcUrl = img.getAttribute("src");
      if (!srcUrl) continue;

      try {
        const response = await fetch(srcUrl);
        if (!response.ok) {
          throw new Error(`HTTP ${response.status} fetching ${srcUrl}`);
        }

        const buffer = await response.arrayBuffer();
        const decryptedBuffer = await decryptBinaryBlob(password, buffer);

        const mimeType = getMimeTypeFromUrl(srcUrl);
        const blob = new Blob([decryptedBuffer], { type: mimeType });
        const blobUrl = URL.createObjectURL(blob);

        img.src = blobUrl;
        img.removeAttribute("data-encrypted-media");
      } catch (mediaErr) {
        console.error(`[Kykloi] Failed to decrypt blob (${srcUrl}):`, mediaErr);
        img.alt = `[Decryption error: ${img.alt || "image"}]`;
      }
    }
  }

  form.addEventListener("submit", async (e) => {
    e.preventDefault();

    if (errorMsg) {

      errorMsg.textContent =

    }

    const password = passwordInput.value;
    const rawPayload = payloadEl.textContent;

    if (submitButton) {
      submitButton.disabled = true;
      submitButton.textContent = "Decrypting...";
    }
    if (passwordInput) {
      passwordInput.disabled = true;
    }

      contentContainer.innerHTML = html;
      contentContainer.style.display = "block";

      payloadEl.textContent = "";
      if (lockContainer) {
        lockContainer.remove();

        errorMsg.style.display = "block";

      if (passwordInput) {
        passwordInput.disabled = false;
        passwordInput.value = "";
        passwordInput.focus();

      if (submitButton) {

        submitButton.textContent = originalButtonText;
      }
```

```
/* search_worker.js - Kykloi SQLite3 WASM FTS5 Search Worker */
"use strict";

importScripts("sqlite3.js");

let sqlite3 = null;
let db = null;
let initPromise = null;

/**
 * Initializes the SQLite3 WASM runtime and deserializes search.db into memory.
 */
async function initDatabase(dbUrl) {
  if (db) return;

  if (!sqlite3) {
    sqlite3 = await self.sqlite3InitModule({
      print: (msg) => console.log("[Kykloi SQLite]", msg),
      printErr: (msg) => console.error("[Kykloi SQLite Error]", msg),
    });
  }

  const response = await fetch(dbUrl);
  if (!response.ok) {
    throw new Error(`HTTP ${response.status} fetching search index from ${dbUrl}`);
  }

  const arrayBuffer = await response.arrayBuffer();
  const bytes = new Uint8Array(arrayBuffer);

  const pData = sqlite3.wasm.allocFromTypedArray(bytes);
  db = new sqlite3.oo1.DB();

  const SQLITE_DESERIALIZE_FREEONCLOSE = sqlite3.capi.SQLITE_DESERIALIZE_FREEONCLOSE || 1;
  const SQLITE_DESERIALIZE_RESIZEABLE = sqlite3.capi.SQLITE_DESERIALIZE_RESIZEABLE || 2;

  const rc = sqlite3.capi.sqlite3_deserialize(
    db.pointer,
    "main",
    pData,
    bytes.byteLength,
    bytes.byteLength,
    SQLITE_DESERIALIZE_FREEONCLOSE | SQLITE_DESERIALIZE_RESIZEABLE
  );

  if (rc !== 0) {
    throw new Error(`sqlite3_deserialize failed with return code: ${rc}`);
  }
}

/**
 * Builds candidate FTS5 queries.
 * Supports advanced syntax (quotes, column filters, boolean operators) with a safe prefix fallback.
 */
function buildFtsQueries(rawQuery) {
  const trimmed = rawQuery.trim();
  if (!trimmed) return [];

  const hasAdvancedSyntax = /["':]|\b(AND|OR|NOT)\b|^-/i.test(trimmed);

  const tokens = trimmed
    .replace(/[^\p{L}\p{N}_]+/gu, " ")
    .trim()
    .split(/\s+/)
    .filter(Boolean);

  const fallbackQuery = tokens.length > 0
    ? tokens.map((t) => `"${t.replace(/"/g, '""')}"*`).join(" ")
    : null;

  if (hasAdvancedSyntax) {
    return [trimmed, fallbackQuery].filter(Boolean);
  }

  return [fallbackQuery].filter(Boolean);
}

/**
 * Returns the 50 most recent documents (default view for empty query).
 */
function getRecentDocuments(searchId) {
  if (!db) return;

  const sql = `
    SELECT title, url, date_edtf
    FROM docs_fts
    ORDER BY (date_edtf != '') DESC, date_edtf DESC, rowid DESC
    LIMIT 50;
  `;

  const rows = [];
  try {
    db.exec({
      sql,
      rowMode: "object",
      callback: (row) => rows.push(row),
    });

    self.postMessage({
      type: "results",
      query: "",
      searchId,
      hits: rows,
      hasMore: false,
      isDefault: true,
    });
  } catch (err) {
    console.warn("[Kykloi Search Worker] Failed to fetch recent documents:", err);
    self.postMessage({
      type: "results",
      query: "",
      searchId,
      hits: [],
      hasMore: false,
      isDefault: true,
      error: err.message,
    });
  }
}

/**
 * Executes full-text search against the docs_fts virtual table.
 */
function executeSearch(query, searchId) {
  const trimmed = (query || "").trim();

  // If query is empty, return the 50 most recent documents
  if (!trimmed) {
    getRecentDocuments(searchId);
    return;
  }

  const candidateQueries = buildFtsQueries(trimmed);
  if (candidateQueries.length === 0 || !db) {
    self.postMessage({
      type: "results",
      query,
      searchId,
      hits: [],
      hasMore: false,
      isDefault: false,
    });
    return;
  }

  const sql = `
    SELECT title, url, date_edtf
    FROM docs_fts
    WHERE docs_fts MATCH ?
    ORDER BY rank
    LIMIT 101;
  `;

  let rows = [];
  let executionSuccess = false;

  for (const ftsQuery of candidateQueries) {
    try {
      rows = [];
      db.exec({
        sql,
        bind: [ftsQuery],
        rowMode: "object",
        callback: (row) => rows.push(row),
      });
      executionSuccess = true;
      break;
    } catch (err) {
      // Loop falls back to the safe prefix query if advanced syntax threw a syntax error
    }
  }

  if (!executionSuccess) {
    self.postMessage({
      type: "results",
      query,
      searchId,
      hits: [],
      hasMore: false,
      isDefault: false,
    });
    return;
  }

  const hasMore = rows.length > 100;
  const hits = hasMore ? rows.slice(0, 100) : rows;

  self.postMessage({
    type: "results",
    query,
    searchId,
    hits,
    hasMore,
    isDefault: false,
  });
}

self.onmessage = async (e) => {
  const { type, dbUrl, query, searchId } = e.data || {};

  switch (type) {
    case "init":
      if (!initPromise) {
        initPromise = initDatabase(dbUrl)
          .then(() => {
            self.postMessage({ type: "ready" });
          })
          .catch((err) => {
            initPromise = null;
            self.postMessage({ type: "error", error: err.message });
          });
      }
      break;

    case "search":
      if (initPromise) {
        try {
          await initPromise;
          executeSearch(query || "", searchId);
        } catch (err) {
          self.postMessage({
            type: "results",
            query,
            searchId,
            hits: [],
            hasMore: false,
            isDefault: false,
            error: err.message,
          });
        }
      }
      break;

    default:
      console.warn("[Kykloi Search Worker] Unhandled message type:", type);
  }
};
```

```
def _ghash_mult(x: int, y: int) -> int:
    r = 0xE1000000000000000000000000000000
    z = 0
    v = x
    for i in range(128):
        if (y >> (127 - i)) & 1:
            z ^= v
        if v & 1:
            v = (v >> 1) ^ r
        else:
            v >>= 1
    return z


def _init_ghash_tables(h: int) -> list[list[int]]:
    tables: list[list[int]] = []
    for j in range(16):
        shift = 8 * (15 - j)
        basis = [_ghash_mult((1 << bit) << shift, h) for bit in range(8)]
        table = [0] * 256
        for b in range(1, 256):
            low_bit = (b & -b).bit_length() - 1
            table[b] = table[b & (b - 1)] ^ basis[low_bit]
        tables.append(table)
    return tables

def pure_aes_gcm_encrypt(
    key: bytes, iv: bytes, plaintext: bytes
) -> tuple[bytes, bytes]:
    if len(key) != 32:
        raise ValueError("Key must be 32 bytes (256 bits).")
    if len(iv) != 12:
        raise ValueError("Initialization vector must be 12 bytes (96 bits).")

    round_keys = _key_expansion_256(key)
    h_bytes = _cipher_block([0] * 16, round_keys)
    h = int.from_bytes(h_bytes, byteorder="big")
    ghash_tables = _init_ghash_tables(h)

    j0 = iv + b"\x00\x00\x00\x01"
    ciphertext = bytearray()
    num_blocks = (len(plaintext) + 15) // 16

    for i in range(num_blocks):
        ctr_val = (1 + 1 + i) & 0xFFFFFFFF
        ctr_block = iv + ctr_val.to_bytes(4, byteorder="big")
        keystream = _cipher_block(list(ctr_block), round_keys)
        chunk = plaintext[i * 16 : (i + 1) * 16]
        for p_b, k_b in zip(chunk, keystream):
            ciphertext.append(p_b ^ k_b)

    ciphertext_bytes = bytes(ciphertext)
    pad_len = (16 - (len(ciphertext_bytes) % 16)) % 16
    padded_ct = ciphertext_bytes + (b"\x00" * pad_len)

    s = 0
    for i in range(0, len(padded_ct), 16):
        block_int = int.from_bytes(padded_ct[i : i + 16], byteorder="big")
        s = _ghash_block(ghash_tables, s ^ block_int)

    len_block_int = len(ciphertext_bytes) * 8
    s = _ghash_block(ghash_tables, s ^ len_block_int)

    tag_mask = _cipher_block(list(j0), round_keys)
    tag = bytes(a ^ b for a, b in zip(s.to_bytes(16, "big"), tag_mask))

    return ciphertext_bytes, tag

def compute_file_hash(path: Path) -> str:
    hasher = hashlib.sha256()
    try:
        with open(path, "rb") as f:
            while chunk := f.read(65536):
                hasher.update(chunk)
        return hasher.hexdigest()
    except (FileNotFoundError, PermissionError) as e:
        logging.warning("Unable to hash %s: %s", path, e)
        return ""

def parse_frontmatter_robust(text: str) -> dict:
    if not text.startswith("---"):
        return {}

    lines = text.splitlines()
    yaml_lines = []
    for line in lines[1:]:
        if line.strip() in ("---", "..."):
            break
        yaml_lines.append(line)

    result = {}
    current_key = None
    multiline_mode = None
    multiline_accumulator = []

    def commit_multiline():
        nonlocal current_key, multiline_mode, multiline_accumulator
        if current_key and multiline_mode:
            if multiline_mode == "|":
                val = "\n".join(multiline_accumulator)
            else:
                val = " ".join(" ".join(multiline_accumulator).split())
            result[current_key] = val
        multiline_mode = None
        multiline_accumulator = []

    for line in yaml_lines:
        stripped = line.strip()
        if not stripped or stripped.startswith("#"):
            continue

        indent = len(line) - len(line.lstrip())

        if multiline_mode and indent > 0:
            multiline_accumulator.append(stripped)
            continue
        elif multiline_mode and indent == 0:
            commit_multiline()

        if stripped.startswith("- ") and current_key:
            item = stripped[2:].strip().strip("'\"")
            if not isinstance(result.get(current_key), list):
                result[current_key] = []
            result[current_key].append(item)
            continue

        if ":" in stripped:
            key, raw_val = stripped.split(":", 1)
            key = key.strip()
            raw_val = raw_val.strip()
            current_key = key

            if raw_val in ("|", ">"):
                multiline_mode = raw_val
                multiline_accumulator = []
                continue

            if raw_val.startswith("[") and raw_val.endswith("]"):
                inner = raw_val[1:-1].strip()
                result[key] = (
                    [x.strip().strip("'\"") for x in inner.split(",") if x.strip()]
                    if inner
                    else []
                )
                continue

            if (raw_val.startswith('"') and raw_val.endswith('"')) or (
                raw_val.startswith("'") and raw_val.endswith("'")
            ):
                raw_val = raw_val[1:-1]

            result[key] = raw_val

    commit_multiline()
    return result

    try:
        with open(md_path, "r", encoding="utf-8") as f:
            header = f.read(16384)

        parsed = parse_frontmatter_robust(header)
        meta["title"] = str(parsed.get("title", "")).strip()
        meta["date_edtf"] = str(
            parsed.get("date_edtf") or parsed.get("date", "")
        ).strip()

        keywords_val = parsed.get("keywords", "")
        if isinstance(keywords_val, list):
            meta["keywords"] = " ".join(str(k).strip() for k in keywords_val)
        elif isinstance(keywords_val, str):
            meta["keywords"] = keywords_val.strip()

    except OSError as err:
        logging.warning("Error reading %s: %s", md_path.name, err)

    if not meta["title"]:
        clean_stem = md_path.stem
        parts = clean_stem.split(maxsplit=1)
        if len(parts) > 1 and parts[0].isdigit() and len(parts[0]) in (8, 12, 14):
            clean_stem = parts[1]
        meta["title"] = clean_stem.replace("-", " ").replace("_", " ").title()

    return md_path, meta


def build_sqlite_search_index(
    records: list[dict], out_path: Path, dry_run: bool
) -> None:
    if dry_run:
        logging.info("[DRY RUN] Would compile SQLite FTS5 search index to %s", out_path)
        return

    out_path.parent.mkdir(parents=True, exist_ok=True)
    tmp_path = out_path.with_suffix(".tmp.db")
    if tmp_path.exists():
        tmp_path.unlink()

    try:
        conn = sqlite3.connect(tmp_path)
        cursor = conn.cursor()

        cursor.execute("PRAGMA page_size = 4096;")
        cursor.execute("PRAGMA journal_mode = OFF;")
        cursor.execute("PRAGMA synchronous = OFF;")

        cursor.execute(
            """
            CREATE VIRTUAL TABLE docs_fts USING fts5(
                title,
                keywords,
                date_edtf,
                url UNINDEXED,
                tokenize='porter unicode61'
            );
            """
        )

        cursor.executemany(
            """
            INSERT INTO docs_fts (title, keywords, date_edtf, url)
            VALUES (?, ?, ?, ?);
            """,
            [
                (
                    r.get("title", ""),
                    r.get("keywords", ""),
                    r.get("date_edtf", ""),
                    r.get("url", ""),
                )
                for r in records
            ],
        )

        conn.commit()
        cursor.execute("INSERT INTO docs_fts(docs_fts) VALUES('optimize');")
        conn.commit()
        cursor.execute("VACUUM;")
        conn.close()

        tmp_path.replace(out_path)
        logging.info(
            "Compiled SQLite FTS5 index to %s (%d records, %d bytes).",
            out_path.name,
            len(records),
            out_path.stat().st_size,
        )

    except Exception as exc:
        if tmp_path.exists():
            tmp_path.unlink()
        logging.error("Failed to build SQLite search index: %s", exc)
        raise

def compile_markdown_worker(task: CompileTask) -> tuple[Path, bool, Optional[str]]:
    if task.dry_run:
        return task.md_path, True, None

    css_files = ["assets/css/pico.css", "assets/css/style.css"]

    # CASE 1: Standard Unencrypted Document
    if not task.is_secure:
        cmd = [
            "pandoc",
            str(task.md_path),
            "--defaults",
            str(task.defaults_yaml),
            "--template",
            str(task.template_path),
            "--lua-filter",
            str(task.lua_filter),
            "-V",
            f"relative_root={task.relative_root}",
            f"--katex={task.relative_root}assets/katex/",
            "-o",
            str(task.out_path),
        ]
        for css in css_files:
            cmd.extend(["--css", f"{task.relative_root}{css}"])

        task.out_path.parent.mkdir(parents=True, exist_ok=True)
        result = subprocess.run(cmd, capture_output=True, text=True)
        if result.returncode != 0:
            return task.md_path, False, result.stderr.strip()
        return task.md_path, True, None

    # CASE 2: Secure Encrypted Document
    if not task.passphrase:
        return (
            task.md_path,
            False,
            "Cannot compile secure file: No passphrase provided.",
        )

    env = os.environ.copy()
    env["KYKLOI_SECURE_BUILD"] = "1"

    body_cmd = [
        "pandoc",
        str(task.md_path),
        "--defaults",
        str(task.defaults_yaml),
        "--lua-filter",
        str(task.lua_filter),
        f"--katex={task.relative_root}assets/katex/",
    ]
    body_result = subprocess.run(body_cmd, capture_output=True, text=True, env=env)
    if body_result.returncode != 0:
        return task.md_path, False, f"Body error: {body_result.stderr.strip()}"

    try:
        ciphertext_b64 = encrypt_payload(body_result.stdout, task.passphrase)
    except Exception as e:
        return task.md_path, False, f"Encryption failed: {e}"

    envelope_cmd = [
        "pandoc",
        str(task.md_path),
        "--defaults",
        str(task.defaults_yaml),
        "--template",
        str(task.template_path),
        "-V",
        f"relative_root={task.relative_root}",
        "-V",
        "encrypted=true",
    ]
    for css in css_files:
        envelope_cmd.extend(["--css", f"{task.relative_root}{css}"])

    envelope_result = subprocess.run(envelope_cmd, capture_output=True, text=True)
    if envelope_result.returncode != 0:
        return (
            task.md_path,
            False,
            f"Envelope error: {envelope_result.stderr.strip()}",
        )

    try:
        final_html = envelope_result.stdout.replace(
            "__KYKLOI_ENCRYPTED_PAYLOAD__", ciphertext_b64
        )
        task.out_path.parent.mkdir(parents=True, exist_ok=True)
        with open(task.out_path, "w", encoding="utf-8") as f:
            f.write(final_html)
    except OSError as e:
        return task.md_path, False, f"Disk write error: {e}"

    return task.md_path, True, None

def prune_orphans(dst_dir: Path, expected_files: set[Path], dry_run: bool) -> int:
    if not dst_dir.exists():
        return 0

    orphans_removed = 0
    for item in list(dst_dir.rglob("*")):
        if item.is_file() and item.resolve() not in expected_files:
            if not dry_run:
                item.unlink()
            orphans_removed += 1
            logging.debug("Pruned orphan file: %s", item)

    for dirpath in sorted(dst_dir.rglob("*"), key=lambda p: len(p.parts), reverse=True):
        if dirpath.is_dir() and not any(dirpath.iterdir()):
            if not dry_run:
                dirpath.rmdir()
            logging.debug("Pruned empty dir: %s", dirpath)

    return orphans_removed
```

```
#!/usr/bin/env python3
"""
migrate_wikilinks.py - High-Performance CommonMark Link Normalizer.
Converts Obsidian, Zettlr, Logseq, and Foam wikilinks into standards-compliant CommonMark.
Handles spaces, anchors, block IDs, frontmatter IDs, and Zettelkasten timestamp prefixes.
Protects KaTeX blocks, code fences, inline code, and YAML frontmatter.
Filters all editor caches, workspaces, OS trash, and swap files.
Part of Project Kykloi.
"""

import argparse
import os
import re
import urllib.parse
from pathlib import Path

# Matches standard Zettelkasten timestamp formats (8 to 14 digits)
ID_PREFIX_PATTERN = re.compile(r"^(\d{14}|\d{12}|\d{8})(?:[-_\s].*)?$")

# Matches Zettlr frontmatter ID: "id: 20240101120000" or "id: '20240101120000'"
FRONTMATTER_ID_PATTERN = re.compile(r"^id:\s*['\"]?(\d{8,14})['\"]?", re.MULTILINE)

# Match ![[target#anchor|alt]] or ![[target]]
WIKI_IMAGE_PATTERN = re.compile(r"!\[\[\s*([^\]\|\n]+?)(?:\|([^\]\n]*))?\s*\]\]")

# Match [[target#anchor|label]] or [[target]]
WIKI_LINK_PATTERN = re.compile(r"\[\[\s*([^\]\|\n]+?)(?:\|([^\]\n]*))?\s*\]\]")

# Carve out protected syntactic blocks from Markdown content
PROTECTED_ZONES_PATTERN = re.compile(
    r"("
    r"^---[\s\S]*?^---|"  # YAML Frontmatter
    r"```[\s\S]*?```|"  # Fenced code blocks (backticks)
    r"~~~[\s\S]*?~~~|"  # Fenced code blocks (tildes)
    r"\$\$(?:\\.|[^\\])*?\$\$|"  # Display math $$ ... $$
    r"\\\[[\s\S]*?\\\]|"  # Display math \[ ... \]
    r"`[^`\n]+`|"  # Inline code
    r"\$(?:\\.|[^\$\\\n])+\$|"  # Inline math $ ... $
    r"\\\([\s\S]*?\\\)"  # Inline math \( ... \)
    r")",
    re.MULTILINE,
)

# Disallowed directory names (exact match, case-insensitive)
JUNK_DIRS = {
    ".obsidian",
    ".trash",
    ".zettlr",
    ".logseq",
    "logseq",
    ".git",
    ".vscode",
    ".idea",
    ".foam",
    ".dendron",
    ".sync",
    "node_modules",
}

# Disallowed filenames (exact match, case-insensitive)
JUNK_EXACT_FILES = {
    ".ds_store",
    "thumbs.db",
    "desktop.ini",
    "ehthumbs.db",
    ".gitignore",
    ".gitattributes",
    ".ztr-directory",
}


def is_junk_path(path: Path) -> bool:
    """
    Determines if a given path is an OS artifact, editor workspace file,
    VCS metadata, lock/swap file, or deleted note in .trash/.
    """
    # Check directory parts for blacklisted directories or hidden directory prefixes
    for part in path.parts[:-1]:
        part_lower = part.lower()
        if part_lower in JUNK_DIRS or part.startswith("."):
            return True

    name = path.name
    name_lower = name.lower()

    # Exact filename match
    if name_lower in JUNK_EXACT_FILES:
        return True

    # Hidden files and OS temporary artifacts
    if name.startswith(".") or name.startswith("._"):
        return True

    # Backup, swap, and temporary editor lock files
    if (
        name.endswith("~")
        or name.endswith(".swp")
        or name.endswith(".swo")
        or name.startswith("~$")
        or (name.startswith(".#") and name.endswith("#"))
    ):
        return True

    return False


def extract_frontmatter_id(path: Path) -> str | None:
    """Fast probe to extract Zettlr/Obsidian ID from YAML header."""
    try:
        with open(path, "r", encoding="utf-8", errors="ignore") as f:
            chunk = f.read(1024)
        if chunk.startswith("---"):
            end_idx = chunk.find("---", 3)
            if end_idx != -1:
                header = chunk[3:end_idx]
                match = FRONTMATTER_ID_PATTERN.search(header)
                if match:
                    return match.group(1)
    except OSError:
        pass
    return None


def build_inventory(src_root: Path) -> dict[str, Path]:
    """
    Constructs an exhaustive dictionary of target lookups from clean files:
    - Relative POSIX path without extension: 'folder/Note'
    - Exact filename: 'Note.md', 'Image.png'
    - File stem (case-sensitive and lowercase): 'Note'
    - Zettelkasten ID (from filename prefix or frontmatter)
    """
    inventory: dict[str, Path] = {}

    for p in src_root.rglob("*"):
        if not p.is_file() or is_junk_path(p):
            continue

        rel = p.relative_to(src_root)
        posix_rel = rel.as_posix()
        name = p.name
        stem = p.stem

        inventory[posix_rel] = rel
        inventory[name] = rel
        inventory[stem] = rel
        inventory[stem.lower()] = rel

        if p.suffix.lower() == ".md":
            no_ext_rel = rel.with_suffix("").as_posix()
            inventory[no_ext_rel] = rel
            inventory[no_ext_rel.lower()] = rel

            # Filename-based ZK ID (e.g., 20240101120000-topic.md)
            m = ID_PREFIX_PATTERN.match(stem)
            if m:
                inventory[m.group(1)] = rel

            # Frontmatter-based ID
            f_id = extract_frontmatter_id(p)
            if f_id:
                inventory[f_id] = rel

    return inventory


def format_commonmark_url(source_rel: Path, target_rel: Path, anchor: str = "") -> str:
    """
    Computes a relative path from source to target, percent-encoding spaces
    and illegal characters while preserving valid path separators and anchor hashes.
    """
    source_dir = source_rel.parent
    rel_path_str = os.path.relpath(target_rel, start=source_dir)
    posix_path = Path(rel_path_str).as_posix()

    encoded_path = urllib.parse.quote(posix_path, safe="/#%")

    if anchor:
        clean_anchor = anchor.strip()
        if not clean_anchor.startswith("#"):
            clean_anchor = "#" + clean_anchor
        encoded_anchor = urllib.parse.quote(clean_anchor, safe="#-_")
        return f"{encoded_path}{encoded_anchor}"

    return encoded_path


def resolve_target(
    raw_target: str, inventory: dict[str, Path]
) -> tuple[Path | None, str]:
    """
    Resolves targets containing anchors, paths, or ZK IDs against clean inventory.
    """
    target = raw_target.strip()
    anchor = ""

    if "#" in target:
        target, anchor = target.split("#", 1)
        target = target.strip()
        anchor = "#" + anchor.strip()

    if not target:
        return None, anchor

    # 1. Direct dictionary match
    if target in inventory:
        return inventory[target], anchor

    # 2. Case-insensitive lookup
    target_lower = target.lower()
    if target_lower in inventory:
        return inventory[target_lower], anchor

    # 3. Basename lookup
    base_name = Path(target).name
    if base_name in inventory:
        return inventory[base_name], anchor

    base_lower = base_name.lower()
    if base_lower in inventory:
        return inventory[base_lower], anchor

    # 4. Leading Zettelkasten ID probe
    zk_match = ID_PREFIX_PATTERN.match(base_name)
    if zk_match:
        zk_id = zk_match.group(1)
        if zk_id in inventory:
            return inventory[zk_id], anchor

    return None, anchor


def convert_file_content(
    doc_rel: Path,
    content: str,
    inventory: dict[str, Path],
) -> tuple[str, int]:
    replacements = 0

    def replace_image(match: re.Match) -> str:
        nonlocal replacements
        raw_target = match.group(1).strip()
        alt_text = match.group(2).strip() if match.group(2) else ""

        target_file, _ = resolve_target(raw_target, inventory)
        if target_file:
            encoded_url = format_commonmark_url(doc_rel, target_file)
            replacements += 1
            display = alt_text or target_file.name
            return f"![{display}]({encoded_url})"

        return match.group(0)

    def replace_link(match: re.Match) -> str:
        nonlocal replacements
        raw_target = match.group(1).strip()
        explicit_label = match.group(2).strip() if match.group(2) else ""

        target_file, anchor = resolve_target(raw_target, inventory)

        if target_file:
            encoded_url = format_commonmark_url(doc_rel, target_file, anchor)
            replacements += 1

            if explicit_label:
                display = explicit_label
            else:
                display = target_file.stem.replace("-", " ").replace("_", " ").title()
                if anchor and not target_file.name:
                    display = anchor.lstrip("#")

            return f"[{display}]({encoded_url})"

        if not target_file and anchor:
            replacements += 1
            display = explicit_label or anchor.lstrip("#")
            clean_anchor = urllib.parse.quote(anchor, safe="#-_")
            return f"[{display}]({clean_anchor})"

        return match.group(0)

    tokens = PROTECTED_ZONES_PATTERN.split(content)
    output_segments: list[str] = []

    for idx, segment in enumerate(tokens):
        if idx % 2 == 1:
            output_segments.append(segment)
        else:
            subbed = WIKI_IMAGE_PATTERN.sub(replace_image, segment)
            subbed = WIKI_LINK_PATTERN.sub(replace_link, subbed)
            output_segments.append(subbed)

    return "".join(output_segments), replacements


def run_migration_pipeline(
    src_root: Path, dry_run: bool = False, verbose: bool = False
) -> tuple[int, int]:
    if not src_root.exists():
        return 0, 0

    md_candidates: list[Path] = []
    for p in src_root.rglob("*.md"):
        if is_junk_path(p):
            continue
        try:
            with open(p, "rb") as f:
                if b"[[" in f.read():
                    md_candidates.append(p)
        except OSError:
            continue

    if not md_candidates:
        return 0, 0

    inventory = build_inventory(src_root)
    files_modified = 0
    links_converted = 0

    for md in md_candidates:
        try:
            with open(md, "r", encoding="utf-8") as f:
                content = f.read()
        except OSError:
            continue

        doc_rel = md.relative_to(src_root)
        new_content, count = convert_file_content(doc_rel, content, inventory)

        if count > 0:
            files_modified += 1
            links_converted += count
            if not dry_run:
                with open(md, "w", encoding="utf-8") as f:
                    f.write(new_content)
            if verbose:
                print(f"[Migrate] Converted {count} link(s) in {doc_rel}")

    return files_modified, links_converted


def main():
    parser = argparse.ArgumentParser(
        description="Migrate Obsidian/Zettlr wikilinks to standard CommonMark relative links."
    )
    parser.add_argument(
        "--dry-run", action="store_true", help="Preview migrations without disk writes."
    )
    parser.add_argument(
        "--verbose", action="store_true", help="Print granular file conversions."
    )
    args = parser.parse_args()

    project_root = Path(__file__).resolve().parent.parent
    src_root = project_root / "content"

    modified, count = run_migration_pipeline(
        src_root=src_root, dry_run=args.dry_run, verbose=args.verbose
    )
    action = "Identified" if args.dry_run else "Converted"
    print(f"Wikilink Migration: {action} {count} link(s) across {modified} file(s).")


if __name__ == "__main__":
    main()
```

## `translate_links.lua`

```
-- translate_links.lua
-- 1. Rewrites relative markdown links (.md) to HTML (.html)
-- 2. If KYKLOI_SECURE_BUILD is active, rewrites image sources to append .enc

local is_secure = os.getenv("KYKLOI_SECURE_BUILD") == "1"

function Link(el)
  -- Skip absolute URLs or protocols
  if el.target:match("^%a+://") or el.target:match("^//") or el.target:match("^mailto:") then
    return el
  end

  -- Decode URL to inspect extension if encoded
  local target = el.target

  -- Separate path from anchor or query
  local path, rest = target:match("^(.-)%.md([#?].*)$")
  if not path then
    path = target:match("^(.-)%.md$")
    rest = ""
  end

  if path then
    el.target = path .. ".html" .. rest
  end

  return el
end

function Image(el)
  if not is_secure then
    return el
  end

  -- Skip absolute URLs, protocols, or existing data URIs
  if el.src:match("^%a+://") or el.src:match("^//") or el.src:match("^data:") then
    return el
  end

  -- Extract clean base URL ignoring anchors or query parameters
  local base_src, suffix = el.src:match("^(.-)(%?.*)$")
  if not base_src then
    base_src = el.src
    suffix = ""
  end

  -- Append .enc if not already present
  if not base_src:match("%.enc$") then
    el.src = base_src .. ".enc" .. suffix
  end

  el.attributes["data-encrypted-media"] = "true"
  return el
end
```
